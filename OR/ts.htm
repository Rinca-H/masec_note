型注釈

型エイリアス

type StringOrNumber = string | number; let value: StringOrNumber; value = "hello"; // string型が代入可能 value = 123; // number型も代入可能

型注釈の書き方

配列

let numbers: number[]; let strings: Array<string>;

読み取り専用配列

const numbers: readonly number[] = [1, 2, 3]; const strings: ReadonlyArray<string> = ["hello", "world"];

タプル型

let tuple: [string, number];

オブジェクト

let a: { name: string; age: number }; let b: { readonly name: string; age: number }; let c: { name: string; age?: number }; let d: { [key: string]: number };

Map

let people: Map<string, number>;

Set

let numSet: Set<number>;

ユニオン型

let value: boolean | number; value = true; // 代入できる value = 100; // 代入できる

インターセクション型

type Octopus = { swims: boolean }; type Cat = { nightVision: boolean }; type Octocat = Octopus & Cat; const octocat: Octocat = { swims: true, nightVision: true }; console.log(octocat);

列挙型

enum Color { Red, Green, Blue, }

値を設定

enum Color { Red = "red", Green = "green", Blue = "blue", }

値にアクセスする場合

const myColor: Color = Color.Red;

型ガード関数

function isString(value: any): value is string { return typeof value === "string"; } function printLength(value: any) { if (isString(value)) { // この節ではvalueはstring型として扱われる console.log(value.length); } } printLength("hello");

ジェネリクス

// Tが型変数 function identity<T>(arg: T): T { return arg; } // 型変数Tにstringを割り当てる const output1 = identity<string>("myString"); // 型変数Tにnumberを割り当てる const output2 = identity<number>(100);

型レベルプログラミング

typeof型演算子

const object = { name: "TypeScript", version: 3.9, }; type ObjectType = typeof object;

keyof型演算子

type Point = { x: number; y: number; }; type Key = keyof Point; const key1: Key = "x"; // 代入OK const key2: Key = "y"; // 代入OK const key3: Key = "z"; // 代入不可

ユーティリティ型

Required

type Person = { name: string; age?: number; }; type RequiredPerson = Required<Person>; // ageがオプションでなくなっている点に注目

Partial

type Person = { name: string; age: number; }; type OptionalPerson = Partial<Person>;

Readonly

type Person = { name: string; age: number; }; type ReadonlyPerson = Readonly<Person>;

Record

type ThreeLetterRecord = Record<"one" | "two" | "three", string>;

Pick

type Person = { name: string; age: number; address: string; }; type PersonNameAndAge = Pick<Person, "name" | "age">;

Omit

type Person = { name: string; age: number; address: string; }; type PersonWithoutAddress = Omit<Person, "address">;

Exclude

type T1 = number | string | boolean; type T2 = Exclude<T1, boolean>;

Extract

type T1 = number | string | boolean; type T2 = string | boolean; type T3 = Extract<T1, T2>;

NonNullable

type T1 = string | null | undefined; type T2 = NonNullable<T1>;

ReturnType

function stringify(value: number): string { return `${value}`; } type StringifyReturnType = ReturnType<typeof stringify>;

Awaited

const promise1 = Promise.resolve("data"); const promise2 = Promise.resolve(Promise.resolve("data")); type Data1 = Awaited<typeof promise1>; type Data2 = Awaited<typeof promise2>;

Mapped types

type Person = { name: string; age: number; }; type ReadOnlyPerson = { readonly [K in keyof Person]: Person[K] };

インデックスアクセス型

type Person = { name: string; age: number; }; type Name = Person["name"];
