概要

言語の特徴

特徴 説明
型付け 静的型付け
パラダイム 主に オブジェクト指向(OOP)、ほかにも手動実行、関数型の要素あり
コンパイル言語 C# は中間言語(IL)にコンパイルされ、.NET ランタイム上で実行
ガベージコレクション メモリ管理は自動で行われる(GCあり)
型安全性 高い (型安全なコードが基本)
マルチパラダイム LINQ による関数型のような記述、非同期処理(async / await)など柔軟な記述が可能

主な用途・得意分野

分野 説明
業務アプリケーション開発 Windows 向けのデスクトップアプリ(WPF、WinForms など)
Webアプリ開発 ASP.NET / ASP.NET Core によるサーバーサイド開発
クラウドアプリ Azure との親和性が高く、クラウドネイティブなアプリ構築に強い
ゲーム開発 Unity（ゲームエンジン）で使用される主要言語
モバイルアプリ開発 .NET MAUI や Xamarin を用いたクロスプラットフォーム開発
IoT・マイクロサービス .NET Core により軽量なアプリケーション開発も可能

async と await

using System; using System.Collections.Generic; using System.Threading; using System.Threading.Tasks; class MyClass { // 同期関数(0から100までを合計した結果を返す同期処理) static int CountUpTask(string name) { int total = 0; for (int i = 0; i <= 100; i++) { total += i; Thread.Sleep(10); Console.Write(name); } return total; } // 非同期関数(上記の関数を非同期関数化したもの) static async Task<int> CountUpAsync(string name) { int total = 0; for (int i = 0; i <= 100; i++) { total += i; await Task.Delay(10); Console.Write(name); } return total; } // awaitを使用するメソッドには必ずasyncをつける static async Task Main() { // 定常処理(ひたすら"."を書き出す) bool stop = false; var taskX = Task.Run(() => { while (!stop) { Console.Write("."); Thread.Sleep(10); } }); // 同期関数をタスクとして非同期的に呼び出す var taskA = Task.Run(() => CountUpTask("A")); Task.WhenAll(taskA); // タスクが完了するのを待つ var countA = taskA.Result; // タスクの結果を取り出す // 非同期関数を同期的に呼び出す(1) // 非同期関数とawaitを使用すれば、上記の処理を同期関数の様に記述することができる var countB = await CountUpAsync("B"); // でも、同期関数を同期的に呼び出すのとどう違うのかを示すサンプルを思いつかない... var countC = CountUpTask("C"); stop = true; Console.WriteLine(" {0} {1} {2}", countA, countB, countC); } }

in / out 修飾子

static void Func(in num) { // num = 999; // 変更しようとするとコンパイルエラーとなる } static void Main() { int num = 123; Func(num); }
static void Func(out int num) { num = 999; // 変更されない場合はエラーとなる } static void Main() { int num = 123; Func(out num); Console.WriteLine(num); // => 999 }

ref 修飾子

static void Func(ref int num) { num = 999; } static void Main() { int num = 123; Func(ref num); Console.WriteLine(num); // => 999 } static int num = 123; static ref int Func() { return ref num; } static void Main() { ref int num = ref Func(); Console.WriteLine(num); }
